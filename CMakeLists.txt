cmake_minimum_required(VERSION 3.20)
project(npplx LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC OFF)

find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets PrintSupport)

# QScintilla (Qt6 variant) -- tiered detection:
# 1. pkg-config (Qt6Qscintilla or qscintilla2_qt6, depending on the distro)
# 2. CMake config package (e.g. Arch: qscintilla-qt6)
# 3. plain library lookup (Debian/Ubuntu: libqscintilla2_qt6,
#    headers live in the Qt6 include path under Qsci/)
find_package(PkgConfig QUIET)
set(NPPLX_QSCI_TARGET "")
if(PkgConfig_FOUND)
  pkg_check_modules(QSCINTILLA QUIET IMPORTED_TARGET Qt6Qscintilla)
  if(NOT TARGET PkgConfig::QSCINTILLA)
    pkg_check_modules(QSCINTILLA QUIET IMPORTED_TARGET qscintilla2_qt6)
  endif()
  if(TARGET PkgConfig::QSCINTILLA)
    set(NPPLX_QSCI_TARGET PkgConfig::QSCINTILLA)
  endif()
endif()
if(NOT NPPLX_QSCI_TARGET)
  find_package(Qt6Qscintilla QUIET)
  if(TARGET Qt6Qscintilla::Qt6Qscintilla)
    set(NPPLX_QSCI_TARGET Qt6Qscintilla::Qt6Qscintilla)
  endif()
endif()
if(NOT NPPLX_QSCI_TARGET)
  find_library(NPPLX_QSCI_LIB NAMES qscintilla2_qt6 qscintilla2-qt6 QScintilla
    DOC "QScintilla library for Qt6")
  if(NOT NPPLX_QSCI_LIB)
    message(FATAL_ERROR "QScintilla for Qt6 not found. Please install the Qt6 development package of QScintilla (Debian/Ubuntu: libqscintilla2-qt6-dev, Arch: qscintilla-qt6, Fedora: qscintilla-qt6-devel).")
  endif()
endif()

add_executable(npplx main.cpp)
target_link_libraries(npplx PRIVATE Qt6::Widgets Qt6::PrintSupport)
if(NPPLX_QSCI_TARGET)
  target_link_libraries(npplx PRIVATE ${NPPLX_QSCI_TARGET})
else()
  target_link_libraries(npplx PRIVATE ${NPPLX_QSCI_LIB})
endif()
